import React, { useState, useRef, useEffect } from 'react';
import { Folder, Mail, User, Briefcase, Gamepad2, FileText } from 'lucide-react';

const MacOSPortfolio = () => {
  const [windows, setWindows] = useState([
    {
      id: 'welcome',
      title: 'Welcome',
      component: 'Welcome',
      isOpen: true,
      isMinimized: false,
      isMaximized: false,
      position: { x: 100, y: 80 },
      size: { width: 600, height: 400 },
      zIndex: 3
    }
  ]);
  const [activeWindow, setActiveWindow] = useState('welcome');
  const [time, setTime] = useState(new Date());
  const [contactForm, setContactForm] = useState({ name: '', email: '', message: '' });
  const [showAboutMenu, setShowAboutMenu] = useState(false);
  const [showViewMenu, setShowViewMenu] = useState(false);
  const [showFolderMenu, setShowFolderMenu] = useState(false);
  const dragRef = useRef({ windowId: null, startX: 0, startY: 0, startWindowX: 0, startWindowY: 0 });

  useEffect(() => {
    const timer = setInterval(() => setTime(new Date()), 1000);
    return () => clearInterval(timer);
  }, []);

  const apps = [
    { id: 'work', title: 'Work', icon: Briefcase, component: 'Work' },
    { id: 'projects', title: 'Projects', icon: Folder, component: 'Projects' },
    { id: 'resume', title: 'Resume', icon: FileText, component: 'Resume' },
    { id: 'games', title: 'Games', icon: Gamepad2, component: 'Games' },
    { id: 'contact', title: 'Contact', icon: Mail, component: 'Contact' }
  ];

  const openApp = (app) => {
    const existingWindow = windows.find(w => w.id === app.id);
    if (existingWindow) {
      if (existingWindow.isMinimized) {
        setWindows(windows.map(w => 
          w.id === app.id ? { ...w, isMinimized: false } : w
        ));
      }
      setActiveWindow(app.id);
      bringToFront(app.id);
    } else {
      const newWindow = {
        id: app.id,
        title: app.title,
        component: app.component,
        isOpen: true,
        isMinimized: false,
        isMaximized: false,
        position: { x: 150 + windows.length * 30, y: 100 + windows.length * 30 },
        size: { width: 700, height: 500 },
        zIndex: Math.max(...windows.map(w => w.zIndex), 0) + 1
      };
      setWindows([...windows, newWindow]);
      setActiveWindow(app.id);
    }
  };

  const closeWindow = (id) => {
    setWindows(windows.filter(w => w.id !== id));
    if (activeWindow === id) {
      const remaining = windows.filter(w => w.id !== id);
      setActiveWindow(remaining.length > 0 ? remaining[remaining.length - 1].id : null);
    }
  };

  const minimizeWindow = (id) => {
    setWindows(windows.map(w => 
      w.id === id ? { ...w, isMinimized: true } : w
    ));
  };

  const maximizeWindow = (id) => {
    setWindows(windows.map(w => 
      w.id === id ? { 
        ...w, 
        isMaximized: !w.isMaximized,
        previousPosition: !w.isMaximized ? w.position : w.previousPosition,
        previousSize: !w.isMaximized ? w.size : w.previousSize
      } : w
    ));
  };

  const bringToFront = (id) => {
    const maxZ = Math.max(...windows.map(w => w.zIndex));
    setWindows(windows.map(w => 
      w.id === id ? { ...w, zIndex: maxZ + 1 } : w
    ));
    setActiveWindow(id);
  };

  const startDrag = (e, windowId) => {
    const win = windows.find(w => w.id === windowId);
    if (win.isMaximized) return;
    
    dragRef.current = {
      windowId,
      startX: e.clientX,
      startY: e.clientY,
      startWindowX: win.position.x,
      startWindowY: win.position.y
    };
    bringToFront(windowId);
  };

  const onDrag = (e) => {
    if (!dragRef.current.windowId) return;
    
    const deltaX = e.clientX - dragRef.current.startX;
    const deltaY = e.clientY - dragRef.current.startY;
    
    setWindows(windows.map(w => 
      w.id === dragRef.current.windowId
        ? {
            ...w,
            position: {
              x: dragRef.current.startWindowX + deltaX,
              y: dragRef.current.startWindowY + deltaY
            }
          }
        : w
    ));
  };

  const stopDrag = () => {
    dragRef.current = { windowId: null, startX: 0, startY: 0, startWindowX: 0, startWindowY: 0 };
  };

  useEffect(() => {
    document.addEventListener('mousemove', onDrag);
    document.addEventListener('mouseup', stopDrag);
    const handleClickOutside = () => {
      setShowAboutMenu(false);
      setShowViewMenu(false);
    };
    document.addEventListener('click', handleClickOutside);
    return () => {
      document.removeEventListener('mousemove', onDrag);
      document.removeEventListener('mouseup', stopDrag);
      document.removeEventListener('click', handleClickOutside);
    };
  }, [windows]);

  const handleContactSubmit = () => {
    alert('Thanks for reaching out! Message received.');
    setContactForm({ name: '', email: '', message: '' });
  };

  const renderWindowContent = (component) => {
    if (component === 'Welcome') {
      return (
        <div className="h-full flex flex-col items-center justify-center text-center p-12 bg-white">
          <div className="text-5xl mb-6 opacity-80">👋</div>
          <h1 className="text-4xl font-light mb-3 text-gray-900 tracking-tight">
            Hakim Nizam
          </h1>
          <p className="text-lg text-gray-500 mb-2 font-light">Design Leader</p>
          <p className="text-sm text-gray-400 max-w-md mb-10 leading-relaxed">
            Creating experiences that bridge people and business success
          </p>
          <button 
            onClick={() => openApp({ id: 'about', title: 'About', component: 'About' })}
            className="px-6 py-2.5 bg-gray-900 text-white text-sm rounded-lg hover:bg-gray-800 transition-colors"
          >
            Explore
          </button>
        </div>
      );
    }
    
    if (component === 'About') {
      return (
        <div className="h-full overflow-auto p-10 bg-white">
          <h2 className="text-2xl font-light mb-8 text-gray-900">About</h2>
          <div className="space-y-6 text-gray-600 leading-relaxed max-w-2xl">
            <p>
              I'm a Kuala Lumpur-based design leader with 5 years of experience taking things apart and figuring out how they tick. That curiosity translates into crafting user experiences that matter.
            </p>
            <p>
              Currently at ServiceRocket, working to become the most reliable Atlassian partner for thousands of businesses worldwide. I also lead the team at Zensite, a top UX agency across Singapore and Malaysia.
            </p>
            <p className="text-sm text-gray-400">
              Active in design leadership communities, staying connected to evolving product trends and debates.
            </p>
            
            <div className="pt-8 space-y-4 border-t border-gray-100 mt-10">
              <h3 className="text-lg font-light text-gray-900 mb-6">Principles</h3>
              <div className="space-y-5">
                <div>
                  <h4 className="text-sm font-medium text-gray-900 mb-1">People First</h4>
                  <p className="text-sm text-gray-500">Understanding needs and motivations is the foundation</p>
                </div>
                <div>
                  <h4 className="text-sm font-medium text-gray-900 mb-1">Vision and Execution</h4>
                  <p className="text-sm text-gray-500">Plan the vision, stay hands-on to bring it to life</p>
                </div>
                <div>
                  <h4 className="text-sm font-medium text-gray-900 mb-1">Fail Fast, Win Big</h4>
                  <p className="text-sm text-gray-500">Learning quickly from failures accelerates success</p>
                </div>
                <div>
                  <h4 className="text-sm font-medium text-gray-900 mb-1">Business Impact</h4>
                  <p className="text-sm text-gray-500">Great design must drive actual results</p>
                </div>
              </div>
            </div>
          </div>
        </div>
      );
    }
    
    if (component === 'Work') {
      return (
        <div className="h-full overflow-auto p-10 bg-white">
          <h2 className="text-2xl font-light mb-8 text-gray-900">Experience</h2>
          <div className="space-y-8 max-w-2xl">
            <div className="pb-8 border-b border-gray-100">
              <h3 className="text-base font-medium text-gray-900 mb-1">ServiceRocket</h3>
              <p className="text-sm text-gray-500 mb-3">Design Leader</p>
              <p className="text-sm text-gray-600 leading-relaxed">
                Leading user experience initiatives for enterprise Atlassian solutions, helping thousands of businesses worldwide work more efficiently.
              </p>
            </div>
            
            <div className="pb-8 border-b border-gray-100">
              <h3 className="text-base font-medium text-gray-900 mb-1">Zensite</h3>
              <p className="text-sm text-gray-500 mb-3">Design Team Lead</p>
              <p className="text-sm text-gray-600 leading-relaxed">
                Guiding a top UX agency across Singapore and Malaysia, delivering exceptional digital experiences for diverse clients.
              </p>
            </div>
            
            <div>
              <h3 className="text-base font-medium text-gray-900 mb-1">Community</h3>
              <p className="text-sm text-gray-500 mb-3">Design Leadership Networks</p>
              <p className="text-sm text-gray-600 leading-relaxed">
                Actively engaged in design communities, staying at the forefront of product design trends and leadership best practices.
              </p>
            </div>
          </div>
        </div>
      );
    }
    
    if (component === 'Projects') {
      return (
        <div className="h-full overflow-auto p-10 bg-white">
          <h2 className="text-2xl font-light mb-8 text-gray-900">Selected Work</h2>
          <div className="grid grid-cols-1 gap-6 max-w-2xl">
            {[1, 2, 3, 4].map(i => (
              <div key={i} className="group cursor-pointer pb-6 border-b border-gray-100 last:border-0">
                <div className="w-full h-48 bg-gray-50 rounded-lg mb-4 group-hover:bg-gray-100 transition-colors"></div>
                <h3 className="text-base font-medium text-gray-900 mb-2">Project {i}</h3>
                <p className="text-sm text-gray-500">Brief description of the project and its impact</p>
              </div>
            ))}
          </div>
        </div>
      );
    }
    
    if (component === 'Resume') {
      return (
        <div className="h-full overflow-auto p-10 bg-white">
          <h2 className="text-2xl font-light mb-8 text-gray-900">Resume</h2>
          <div className="space-y-10 max-w-2xl">
            <section>
              <h3 className="text-sm font-medium text-gray-900 mb-4 uppercase tracking-wider">Experience</h3>
              <div className="space-y-6">
                <div>
                  <h4 className="text-base font-medium text-gray-900">Design Leader</h4>
                  <p className="text-sm text-gray-500 mb-2">ServiceRocket · 2020 - Present</p>
                  <p className="text-sm text-gray-600">Leading UX initiatives for enterprise solutions</p>
                </div>
                <div>
                  <h4 className="text-base font-medium text-gray-900">Design Team Lead</h4>
                  <p className="text-sm text-gray-500 mb-2">Zensite · 2019 - Present</p>
                  <p className="text-sm text-gray-600">Managing UX agency operations across Singapore and Malaysia</p>
                </div>
              </div>
            </section>
            
            <section>
              <h3 className="text-sm font-medium text-gray-900 mb-4 uppercase tracking-wider">Skills</h3>
              <div className="flex flex-wrap gap-2">
                {['UX Design', 'Design Leadership', 'Product Strategy', 'User Research', 'Figma', 'Team Management', 'Atlassian'].map(skill => (
                  <span key={skill} className="px-3 py-1.5 bg-gray-50 text-gray-700 rounded-md text-xs">
                    {skill}
                  </span>
                ))}
              </div>
            </section>
          </div>
        </div>
      );
    }
    
    if (component === 'Games') {
      return (
        <div className="h-full flex flex-col items-center justify-center bg-white p-10">
          <Gamepad2 size={48} className="text-gray-300 mb-6" />
          <h2 className="text-2xl font-light mb-3 text-gray-900">Games</h2>
          <p className="text-center mb-8 text-gray-500 text-sm max-w-md">
            Ready for a break? DOOM and Oregon Trail coming soon.
          </p>
          <div className="space-y-3 w-64">
            <button className="w-full px-6 py-2.5 bg-gray-900 text-white text-sm rounded-lg hover:bg-gray-800 transition-colors">
              DOOM
            </button>
            <button className="w-full px-6 py-2.5 bg-gray-100 text-gray-900 text-sm rounded-lg hover:bg-gray-200 transition-colors">
              Oregon Trail
            </button>
          </div>
        </div>
      );
    }
    
    if (component === 'Contact') {
      return (
        <div className="h-full overflow-auto p-10 bg-white">
          <h2 className="text-2xl font-light mb-8 text-gray-900">Get in touch</h2>
          <div className="space-y-5 max-w-md">
            <div>
              <label className="block text-xs font-medium text-gray-700 mb-2 uppercase tracking-wider">Name</label>
              <input 
                type="text" 
                value={contactForm.name}
                onChange={(e) => setContactForm({...contactForm, name: e.target.value})}
                className="w-full px-4 py-2.5 border border-gray-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-gray-400 focus:border-transparent text-sm" 
              />
            </div>
            <div>
              <label className="block text-xs font-medium text-gray-700 mb-2 uppercase tracking-wider">Email</label>
              <input 
                type="email" 
                value={contactForm.email}
                onChange={(e) => setContactForm({...contactForm, email: e.target.value})}
                className="w-full px-4 py-2.5 border border-gray-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-gray-400 focus:border-transparent text-sm" 
              />
            </div>
            <div>
              <label className="block text-xs font-medium text-gray-700 mb-2 uppercase tracking-wider">Message</label>
              <textarea 
                rows={4} 
                value={contactForm.message}
                onChange={(e) => setContactForm({...contactForm, message: e.target.value})}
                className="w-full px-4 py-2.5 border border-gray-200 rounded-lg focus:outline-none focus:ring-1 focus:ring-gray-400 focus:border-transparent text-sm"
              ></textarea>
            </div>
            <button 
              onClick={handleContactSubmit}
              className="w-full px-6 py-2.5 bg-gray-900 text-white text-sm rounded-lg hover:bg-gray-800 transition-colors"
            >
              Send
            </button>
          </div>
          <div className="mt-10 pt-8 border-t border-gray-100">
            <p className="text-xs text-gray-500 mb-4 uppercase tracking-wider">Connect</p>
            <div className="flex space-x-3">
              <a href="https://www.linkedin.com" target="_blank" rel="noopener noreferrer" className="px-4 py-2 bg-gray-50 text-gray-900 rounded-lg hover:bg-gray-100 text-xs transition-colors">LinkedIn</a>
              <a href="https://dribbble.com" target="_blank" rel="noopener noreferrer" className="px-4 py-2 bg-gray-50 text-gray-900 rounded-lg hover:bg-gray-100 text-xs transition-colors">Dribbble</a>
              <a href="https://www.behance.net" target="_blank" rel="noopener noreferrer" className="px-4 py-2 bg-gray-50 text-gray-900 rounded-lg hover:bg-gray-100 text-xs transition-colors">Behance</a>
            </div>
          </div>
        </div>
      );
    }
    
    return <div className="h-full flex items-center justify-center bg-white">Content loading...</div>;
  };

  return (
    <div className="w-full h-screen bg-gray-50 overflow-hidden relative">
      <div className="absolute inset-0">
        <svg className="w-full h-full" xmlns="http://www.w3.org/2000/svg">
          <defs>
            <linearGradient id="grad1" x1="0%" y1="0%" x2="100%" y2="100%">
              <stop offset="0%" style={{ stopColor: '#f9fafb', stopOpacity: 1 }} />
              <stop offset="50%" style={{ stopColor: '#f3f4f6', stopOpacity: 1 }} />
              <stop offset="100%" style={{ stopColor: '#e5e7eb', stopOpacity: 1 }} />
            </linearGradient>
          </defs>
          <rect width="100%" height="100%" fill="url(#grad1)" />
          <g opacity="0.4">
            <path fill="none" stroke="#d1d5db" strokeWidth="1.5">
              <animate
                attributeName="d"
                dur="20s"
                repeatCount="indefinite"
                values="M0,150 Q400,50 800,150 T1600,150 T2400,150;
                        M0,150 Q400,250 800,150 T1600,150 T2400,150;
                        M0,150 Q400,50 800,150 T1600,150 T2400,150"
              />
            </path>
            <path fill="none" stroke="#d1d5db" strokeWidth="1.5" opacity="0.6">
              <animate
                attributeName="d"
                dur="25s"
                repeatCount="indefinite"
                values="M0,300 Q500,200 1000,300 T2000,300 T3000,300;
                        M0,300 Q500,400 1000,300 T2000,300 T3000,300;
                        M0,300 Q500,200 1000,300 T2000,300 T3000,300"
              />
            </path>
            <path fill="none" stroke="#d1d5db" strokeWidth="1.5" opacity="0.4">
              <animate
                attributeName="d"
                dur="30s"
                repeatCount="indefinite"
                values="M0,450 Q600,350 1200,450 T2400,450 T3600,450;
                        M0,450 Q600,550 1200,450 T2400,450 T3600,450;
                        M0,450 Q600,350 1200,450 T2400,450 T3600,450"
              />
            </path>
          </g>
        </svg>
      </div>

      <div className="absolute top-0 left-0 right-0 h-11 bg-white bg-opacity-60 backdrop-blur-2xl border-b border-gray-200 border-opacity-50 flex items-center px-5 z-50" style={{
        backdropFilter: 'blur(40px) saturate(180%)',
        WebkitBackdropFilter: 'blur(40px) saturate(180%)'
      }}>
        <div className="flex items-center space-x-6 text-sm font-light text-gray-700">
          <span className="font-normal">Hakim OS</span>
          
          <div className="relative">
            <button 
              onClick={(e) => {
                e.stopPropagation();
                setShowAboutMenu(!showAboutMenu);
                setShowViewMenu(false);
              }}
              className="cursor-pointer hover:text-gray-900 transition-colors px-2 py-1 rounded hover:bg-gray-100 hover:bg-opacity-50"
            >
              About
            </button>
            {showAboutMenu && (
              <div className="absolute top-full left-0 mt-1 w-72 bg-white bg-opacity-95 backdrop-blur-2xl rounded-lg shadow-2xl border border-gray-200 border-opacity-50 p-4 z-50"
                style={{
                  backdropFilter: 'blur(40px) saturate(180%)',
                  WebkitBackdropFilter: 'blur(40px) saturate(180%)'
                }}
              >
                <h3 className="text-base font-medium text-gray-900 mb-1">Hakim Nizam</h3>
                <p className="text-sm text-gray-600 mb-2">Design Leader</p>
                <p className="text-xs text-gray-500 leading-relaxed mb-3">
                  Top 1% ADPList Mentor helping designers navigate career growth, portfolio reviews, and leadership transitions. 5 years of experience crafting user experiences at ServiceRocket and Zensite.
                </p>
                <button 
                  onClick={() => {
                    setShowAboutMenu(false);
                    openApp({ id: 'about', title: 'About', component: 'About' });
                  }}
                  className="text-xs text-gray-700 hover:text-gray-900 underline"
                >
                  Learn more...
                </button>
              </div>
            )}
          </div>
          
          <div className="relative">
            <button 
              onClick={(e) => {
                e.stopPropagation();
                setShowViewMenu(!showViewMenu);
                setShowAboutMenu(false);
              }}
              className="cursor-pointer hover:text-gray-900 transition-colors px-2 py-1 rounded hover:bg-gray-100 hover:bg-opacity-50"
            >
              View
            </button>
            {showViewMenu && (
              <div className="absolute top-full left-0 mt-1 w-48 bg-white bg-opacity-95 backdrop-blur-2xl rounded-lg shadow-2xl border border-gray-200 border-opacity-50 py-2 z-50"
                style={{
                  backdropFilter: 'blur(40px) saturate(180%)',
                  WebkitBackdropFilter: 'blur(40px) saturate(180%)'
                }}
              >
                <button 
                  onClick={() => {
                    setShowViewMenu(false);
                    openApp({ id: 'work', title: 'Work', component: 'Work' });
                  }}
                  className="w-full text-left px-4 py-2 text-sm text-gray-700 hover:bg-gray-100 hover:bg-opacity-50 transition-colors"
                >
                  Work Experience
                </button>
                <button 
                  onClick={() => {
                    setShowViewMenu(false);
                    openApp(apps.find(a => a.id === 'projects'));
                  }}
                  className="w-full text-left px-4 py-2 text-sm text-gray-700 hover:bg-gray-100 hover:bg-opacity-50 transition-colors"
                >
                  Projects
                </button>
                <button 
                  onClick={() => {
                    setShowViewMenu(false);
                    openApp(apps.find(a => a.id === 'resume'));
                  }}
                  className="w-full text-left px-4 py-2 text-sm text-gray-700 hover:bg-gray-100 hover:bg-opacity-50 transition-colors"
                >
                  Resume
                </button>
                <div className="h-px bg-gray-200 my-2"></div>
                <button 
                  onClick={() => {
                    setShowViewMenu(false);
                    openApp(apps.find(a => a.id === 'games'));
                  }}
                  className="w-full text-left px-4 py-2 text-sm text-gray-700 hover:bg-gray-100 hover:bg-opacity-50 transition-colors"
                >
                  Games
                </button>
                <button 
                  onClick={() => {
                    setShowViewMenu(false);
                    openApp(apps.find(a => a.id === 'contact'));
                  }}
                  className="w-full text-left px-4 py-2 text-sm text-gray-700 hover:bg-gray-100 hover:bg-opacity-50 transition-colors"
                >
                  Contact
                </button>
              </div>
            )}
          </div>
        </div>
        <div className="ml-auto text-xs font-light text-gray-600">
          {time.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' })}
        </div>
      </div>

      {windows.map(window => (
        !window.isMinimized && (
          <div
            key={window.id}
            className="absolute shadow-2xl overflow-hidden flex flex-col"
            style={{
              left: window.isMaximized ? 0 : window.position.x,
              top: window.isMaximized ? 44 : window.position.y,
              width: window.isMaximized ? '100%' : window.size.width,
              height: window.isMaximized ? 'calc(100% - 116px)' : window.size.height,
              zIndex: window.zIndex,
              borderRadius: window.isMaximized ? '0' : '12px',
              transition: window.isMaximized ? 'all 0.3s cubic-bezier(0.4, 0, 0.2, 1)' : 'none',
              background: 'rgba(255, 255, 255, 0.8)',
              backdropFilter: 'blur(40px) saturate(180%)',
              WebkitBackdropFilter: 'blur(40px) saturate(180%)',
              border: '1px solid rgba(255, 255, 255, 0.3)'
            }}
          >
            <div
              className={`h-11 flex items-center justify-between px-5 cursor-move border-b border-gray-200 border-opacity-30 ${
                activeWindow === window.id ? 'bg-white bg-opacity-40' : 'bg-white bg-opacity-20'
              }`}
              onMouseDown={(e) => startDrag(e, window.id)}
              style={{
                backdropFilter: 'blur(20px)',
                WebkitBackdropFilter: 'blur(20px)'
              }}
            >
              <div className="flex items-center space-x-2">
                <button
                  onClick={() => closeWindow(window.id)}
                  className="w-3 h-3 rounded-full bg-red-400 hover:bg-red-500 transition-colors"
                ></button>
                <button
                  onClick={() => minimizeWindow(window.id)}
                  className="w-3 h-3 rounded-full bg-yellow-400 hover:bg-yellow-500 transition-colors"
                ></button>
                <button
                  onClick={() => maximizeWindow(window.id)}
                  className="w-3 h-3 rounded-full bg-green-400 hover:bg-green-500 transition-colors"
                ></button>
              </div>
              <span className="text-sm font-light text-gray-700">{window.title}</span>
              <div className="w-16"></div>
            </div>
            
            <div className="flex-1 overflow-hidden">
              {renderWindowContent(window.component)}
            </div>
          </div>
        )
      ))}

      <div className="absolute bottom-4 left-1/2 transform -translate-x-1/2 px-3 py-2 z-50" style={{
        background: 'rgba(255, 255, 255, 0.5)',
        backdropFilter: 'blur(40px) saturate(180%)',
        WebkitBackdropFilter: 'blur(40px) saturate(180%)',
        borderRadius: '16px',
        border: '1px solid rgba(255, 255, 255, 0.3)',
        boxShadow: '0 8px 32px rgba(0, 0, 0, 0.08)'
      }}>
        <div className="flex items-center space-x-1.5">
          {apps.map(app => (
            <button
              key={app.id}
              onClick={() => openApp(app)}
              className="w-12 h-12 rounded-xl flex items-center justify-center text-gray-700 hover:bg-white hover:bg-opacity-60 transition-all hover:scale-105"
              title={app.title}
              style={{
                backdropFilter: 'blur(10px)',
                WebkitBackdropFilter: 'blur(10px)'
              }}
            >
              <app.icon size={22} strokeWidth={1.5} />
            </button>
          ))}
        </div>
      </div>
    </div>
  );
};
